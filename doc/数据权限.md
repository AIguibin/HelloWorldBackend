## 两者的区别和职责：

### 1. **sys_role.data_scope_type** - **角色全局默认数据范围**
- **定位**：角色级别的**全局默认**数据权限范围
- **作用**：为角色设置一个**默认的、通用的**数据访问范围
- **场景**：
  - 当系统没有为特定实体配置数据权限规则时，使用此默认范围
  - 为用户快速分配一个基础的数据访问范围
  - 作为数据权限计算的**默认兜底规则**
- **级别**：**角色级**，全局配置

### 2. **sys_role_org.org_range_type** - **角色在特定机构的权限范围**
- **定位**：角色在**特定机构**上的**具体操作权限范围**
- **作用**：定义角色对**某个具体机构**的访问粒度
- **场景**：
  - 超级管理员在"系统管理中心"机构拥有**全部**权限（包含下级）
  - 部门经理在"产品中心"机构拥有**管理**权限
  - 正式访客在"开发中心"机构只有**查看**权限
- **级别**：**角色-机构级**，具体到每个机构的关系配置

## 设计合理性分析：

### 优点：
1. **分离关注点**：
   - `data_scope_type`：控制**数据能看多少**
   - `org_range_type`：控制**操作能干什么**

2. **灵活度提升**：
   - 角色可以设置默认数据范围（如"本人"）
   - 同时对特定机构可以扩展范围（如"包含下级机构"）
   - 实现**最小权限+按需扩展**的安全模型

3. **典型应用场景**：
   ```sql
   -- 角色：系统管理员
   -- data_scope_type: 2 (本机构) -- 默认只能看自己机构数据
   -- 但在 sys_role_org 表中：
   INSERT INTO sys_role_org VALUES ('RL000002', 'JG00000001', 2, 1, ...); -- 对系统管理中心：包含下级机构，管理权限
   INSERT INTO sys_role_org VALUES ('RL000002', 'JG00000002', 1, 2, ...); -- 对产品中心：仅本机构，查看权限
   ```

4. **系统演化路径**：
   - 初期：只使用 `data_scope_type` 简化权限
   - 成熟期：引入 `sys_role_org` 做精细化权限控制

        private Integer orgRangeType;            // 机构范围类型（当数据范围为机构时有效）
        private Integer permType;                // 权限类型（最高权限）
        private Set<String> authorizedOrgCodes;  // 有权限的机构编码集合
        private boolean canManage;               // 是否有管理权限
        private boolean canView;                 // 是否有查看权限
        private boolean canOperate;              // 是否有操作权限
    }

    public DataScopeResult calculateDataScopeWithCache(String userNum, String targetOrgCode) {
        // 构建缓存key
        String cacheKey = String.format("data_scope:%s:%s", userNum, targetOrgCode);
        
        // 尝试从缓存获取
        DataScopeResult cachedResult = getFromCache(cacheKey);
        if (cachedResult != null) {
            return cachedResult;
        }
        
        // 从数据库查询并计算
        DataScopeResult result = calculateDataScope(userNum, targetOrgCode);
        
        // 存入缓存（设置合适的过期时间）
        saveToCache(cacheKey, result, 5 * 60); // 5分钟
        
        return result;
    }
    
    /**
     * 计算用户在目标机构的数据范围权限（主计算方法）
     */
    public DataScopeResult calculateDataScope(String userNum, String targetOrgCode) {
        long startTime = System.currentTimeMillis();
        
        try {
            // 1. 查询用户角色信息（批量查询，避免N+1问题）
            log.debug("开始查询用户[{}]的角色信息", userNum);
            List<Map<String, Object>> userRoles = queryUserRoles(userNum);
            
            if (userRoles.isEmpty()) {
                log.info("用户[{}]没有分配任何角色，使用最小权限", userNum);
                return createMinimalResult();
            }
            
            // 2. 提取角色编码，批量查询角色机构范围
            Set<String> roleCodes = extractRoleCodes(userRoles);
            log.debug("用户[{}]的角色编码: {}", userNum, roleCodes);
            
            List<Map<String, Object>> roleOrgScopes = queryRoleOrgScopes(roleCodes);
            
            // 3. 预计算：构建快速查找的数据结构
            Map<String, List<Map<String, Object>>> orgScopesByRole = preprocessOrgScopes(roleOrgScopes);
            
            // 4. 计算数据范围
            DataScopeResult result = calculate(userRoles, orgScopesByRole, targetOrgCode);
            
            // 5. 计算权限类型
            calculatePermissionTypes(result, userRoles, orgScopesByRole, targetOrgCode);
            
            long costTime = System.currentTimeMillis() - startTime;
            log.debug("用户[{}]在机构[{}]的数据范围计算完成，耗时: {}ms", userNum, targetOrgCode, costTime);
            
            return result;
            
        } catch (Exception e) {
            log.error("计算用户[{}]在机构[{}]的数据范围时发生异常", userNum, targetOrgCode, e);
            // 异常时返回最小权限，保证系统安全
            return createMinimalResult();
        }
    }
    
    /**
     * 核心计算逻辑
     */
    private DataScopeResult calculate(List<Map<String, Object>> userRoles,
                                     Map<String, List<Map<String, Object>>> orgScopesByRole,
                                     String targetOrgCode) {
        
        // 初始化结果为最小权限
        DataScopeResult result = new DataScopeResult();
        result.setDataScopeType(DATA_SCOPE_SELF);
        result.setPermType(PERM_TYPE_VIEW); // 默认查看权限
        result.setAuthorizedOrgCodes(new HashSet<>());
        
        // 用于记录各机构的最高权限类型
        Map<String, Integer> orgPermTypes = new HashMap<>();
        
        // 遍历所有角色
        for (Map<String, Object> role : userRoles) {
            String roleCode = (String) role.get("roleCode");
            Integer roleDataScope = (Integer) role.get("dataScopeType");
            
            // 获取该角色的机构范围配置
            List<Map<String, Object>> roleOrgScopes = orgScopesByRole.getOrDefault(roleCode, Collections.emptyList());
            
            // 计算该角色能贡献的权限
            processRoleContribution(role, roleOrgScopes, result, orgPermTypes, targetOrgCode);
        }
        
        // 处理自定义数据范围
        if (result.getDataScopeType() == DATA_SCOPE_CUSTOM) {
            handleCustomDataScope(result, orgPermTypes, targetOrgCode);
        }
        
        // 设置权限类型标志
        setPermissionFlags(result);
        
        return result;
    }
    
    /**
     * 处理单个角色的权限贡献
     */
    private void processRoleContribution(Map<String, Object> role,
                                        List<Map<String, Object>> roleOrgScopes,
                                        DataScopeResult result,
                                        Map<String, Integer> orgPermTypes,
                                        String targetOrgCode) {
        String roleCode = (String) role.get("roleCode");
        Integer roleDataScope = (Integer) role.get("dataScopeType");
        
        // 1. 更新数据范围（取最宽）
        if (roleDataScope < result.getDataScopeType()) {
            result.setDataScopeType(roleDataScope);
        }
        
        // 2. 处理该角色的机构范围配置
        for (Map<String, Object> scope : roleOrgScopes) {
            String orgCode = (String) scope.get("orgCode");
            Integer orgRangeType = (Integer) scope.get("orgRangeType");
            Integer permType = (Integer) scope.get("permType");
            
            // 记录机构权限
            if (orgPermTypes.containsKey(orgCode)) {
                // 取权限类型中的最小值（1-管理 < 2-查看 < 3-操作）
                int currentPerm = orgPermTypes.get(orgCode);
                orgPermTypes.put(orgCode, Math.min(currentPerm, permType));
            } else {
                orgPermTypes.put(orgCode, permType);
            }
            
            // 添加到授权机构集合
            result.getAuthorizedOrgCodes().add(orgCode);
            
            // 如果是目标机构，更新机构范围类型（取最宽）
            if (targetOrgCode.equals(orgCode)) {
                if (result.getOrgRangeType() == null || orgRangeType < result.getOrgRangeType()) {
                    result.setOrgRangeType(orgRangeType);
                }
                // 更新权限类型（取最高，即最小值）
                if (permType < result.getPermType()) {
                    result.setPermType(permType);
                }
            }
            
            // 处理特殊机构"ALL"
            if (ORG_CODE_ALL.equals(orgCode)) {
                // 如果有全部机构的管理权限，直接升级为全部数据范围
                if (permType == PERM_TYPE_MANAGE && orgRangeType == ORG_RANGE_WITH_CHILDREN) {
                    result.setDataScopeType(DATA_SCOPE_ALL);
                }
            }
        }
    }
    
    /**
     * 处理自定义数据范围
     */
    private void handleCustomDataScope(DataScopeResult result,
                                      Map<String, Integer> orgPermTypes,
                                      String targetOrgCode) {
        // 检查是否有目标机构的权限
        if (result.getAuthorizedOrgCodes().contains(targetOrgCode)) {
            // 有目标机构权限
            if (result.getOrgRangeType() == ORG_RANGE_WITH_CHILDREN) {
                result.setDataScopeType(DATA_SCOPE_ORG_WITH_CHILDREN);
            } else {
                result.setDataScopeType(DATA_SCOPE_ORG);
            }
        } else if (result.getAuthorizedOrgCodes().contains(ORG_CODE_ALL)) {
            // 有全部机构权限
            Integer allOrgRangeType = getOrgRangeTypeForAll(orgPermTypes);
            if (allOrgRangeType == ORG_RANGE_WITH_CHILDREN) {
                result.setDataScopeType(DATA_SCOPE_ALL);
            } else {
                result.setDataScopeType(DATA_SCOPE_ORG);
            }
        } else {
            // 没有特殊权限，保持本人权限
            result.setDataScopeType(DATA_SCOPE_SELF);
        }
    }
    
    /**
     * 获取全部机构的范围类型
     */
    private Integer getOrgRangeTypeForAll(Map<String, Integer> orgPermTypes) {
        // 这里简化处理，实际可能需要更复杂的逻辑
        // 假设ALL机构只有一个配置
        return ORG_RANGE_WITH_CHILDREN; // 默认包含下级
    }
    
    /**
     * 设置权限类型标志
     */
    private void setPermissionFlags(DataScopeResult result) {
        result.setCanManage(result.getPermType() == PERM_TYPE_MANAGE);
        result.setCanView(result.getPermType() <= PERM_TYPE_VIEW);
        result.setCanOperate(result.getPermType() <= PERM_TYPE_OPERATE);
    }

    /**
     * 批量计算多个用户的数据范围（性能优化）
     */
    public Map<String, DataScopeResult> batchCalculateDataScope(List<String> userNums, String targetOrgCode) {
        // 1. 批量查询所有用户的角色
        Map<String, List<Map<String, Object>>> allUserRoles = batchQueryUserRoles(userNums);
        
        // 2. 收集所有角色编码，批量查询机构范围
        Set<String> allRoleCodes = collectAllRoleCodes(allUserRoles);
        Map<String, List<Map<String, Object>>> orgScopesByRole = batchQueryRoleOrgScopes(allRoleCodes);
        
        // 3. 并行计算每个用户的数据范围
        Map<String, DataScopeResult> results = new ConcurrentHashMap<>();
        
        userNums.parallelStream().forEach(userNum -> {
            List<Map<String, Object>> userRoles = allUserRoles.getOrDefault(userNum, Collections.emptyList());
            DataScopeResult result = calculate(userRoles, orgScopesByRole, targetOrgCode);
            results.put(userNum, result);
        });
        
        return results;
    }
    
    /**
     * 批量查询用户角色
     */
    private Map<String, List<Map<String, Object>>> batchQueryUserRoles(List<String> userNums) {
        // 使用IN查询优化
        return sysUserRoleMapper.batchSelectUserRolesWithDataScope(userNums);
    }
    
    /**
     * 批量查询角色机构范围
     */
    private Map<String, List<Map<String, Object>>> batchQueryRoleOrgScopes(Set<String> roleCodes) {
        if (roleCodes.isEmpty()) {
            return Collections.emptyMap();
        }
        
        List<Map<String, Object>> scopes = sysRoleOrgMapper.batchSelectRoleOrgScopes(new ArrayList<>(roleCodes));
        return preprocessOrgScopes(scopes);
    }
    
    /**
     * 收集所有角色编码
     */
    private Set<String> collectAllRoleCodes(Map<String, List<Map<String, Object>>> allUserRoles) {
        Set<String> allRoleCodes = new HashSet<>();
        for (List<Map<String, Object>> roles : allUserRoles.values()) {
            for (Map<String, Object> role : roles) {
                allRoleCodes.add((String) role.get("roleCode"));
            }
        }
        return allRoleCodes;
    }
