# 基于统一权限设计的大一统权限实现思路

## 一、核心设计思想

### 1.1 统一权限模型（Unified Permission Model）
将所有权限抽象为统一的模型，通过 `sys_permission` 表集中管理：

```
权限(perm_code)
├── 元信息 (权限类型、标识、描述)
├── 关联关系 (关联的菜单、数据规则、字段权限、时间权限)
└── 授权配置 (角色授权、生效时间、优先级)
```

### 1.2 权限计算流程
```
用户登录 → 确定机构 → 加载角色 → 查询权限 → 计算数据范围 → 构建权限集合
```

## 二、具体实现思路

### 2.1 统一权限查询Mapper

```java
package com.example.permission.mapper;

import org.apache.ibatis.annotations.Mapper;
import org.apache.ibatis.annotations.Param;
import org.apache.ibatis.annotations.Select;
import java.util.List;
import java.util.Map;

@Mapper
public interface UnifiedPermissionMapper {
    
    /**
     * 查询用户在指定机构下的所有权限（大一统查询）
     * @param userNum 用户编号
     * @param orgCode 机构编码
     * @return 权限信息列表
     */
    @Select("<script>" +
            "SELECT DISTINCT " +
            "  p.perm_code, " +
            "  p.perm_name, " +
            "  p.perm_key, " +
            "  p.perm_type, " +
            "  p.menu_code, " +
            "  p.api_path, " +
            "  p.entity_type, " +
            "  p.entity_field, " +
            "  p.rule_type, " +
            "  p.scope_type, " +
            "  p.custom_rule, " +
            "  p.is_default, " +
            "  rp.auth_type, " +
            "  rp.priority, " +
            "  rp.effective_start, " +
            "  rp.effective_end, " +
            "  m.menu_name, " +
            "  m.menu_type, " +
            "  m.parent_menu_code, " +
            "  m.icon, " +
            "  m.path, " +
            "  m.component, " +
            "  m.url, " +
            "  m.http_method, " +
            "  m.is_external, " +
            "  m.is_cache, " +
            "  m.is_visible, " +
            "  m.sort_order as menu_order " +
            "FROM sys_user_role ur " +
            "JOIN sys_role_permission rp ON ur.role_code = rp.role_code " +
            "JOIN sys_permission p ON rp.perm_code = p.perm_code " +
            "LEFT JOIN sys_menu m ON p.menu_code = m.menu_code " +
            "WHERE ur.user_num = #{userNum} " +
            "  AND ur.status = 1 " +
            "  AND ur.is_deleted = 0 " +
            "  AND (ur.effective_end IS NULL OR ur.effective_end > NOW()) " +
            "  AND rp.status = 1 " +
            "  AND rp.is_deleted = 0 " +
            "  AND (rp.effective_end IS NULL OR rp.effective_end > NOW()) " +
            "  AND p.status = 1 " +
            "  AND p.is_deleted = 0 " +
            "  AND EXISTS ( " +
            "    SELECT 1 FROM sys_user_org uo " +
            "    WHERE uo.user_num = ur.user_num " +
            "      AND uo.org_code = #{orgCode} " +
            "      AND uo.status = 1 " +
            "      AND uo.is_deleted = 0 " +
            "  ) " +
            "ORDER BY p.perm_type, rp.priority, m.sort_order" +
            "</script>")
    List<Map<String, Object>> selectAllPermissionsByUserAndOrg(
            @Param("userNum") String userNum,
            @Param("orgCode") String orgCode);
    
    /**
     * 查询权限关联的数据规则
     * @param permCodes 权限编码列表
     * @return 数据规则列表
     */
    @Select("<script>" +
            "SELECT " +
            "  dr.rule_code, " +
            "  dr.rule_name, " +
            "  dr.rule_type, " +
            "  dr.perm_code, " +
            "  dr.entity_type, " +
            "  dr.scope_type, " +
            "  dr.include_children, " +
            "  dr.custom_sql, " +
            "  dr.rule_expression, " +
            "  dr.rule_priority, " +
            "  dr.is_global " +
            "FROM sys_data_rule dr " +
            "WHERE dr.perm_code IN " +
            "  <foreach collection='permCodes' item='permCode' open='(' separator=',' close=')'>" +
            "    #{permCode}" +
            "  </foreach>" +
            "  AND dr.status = 1 " +
            "  AND dr.is_deleted = 0")
    List<Map<String, Object>> selectDataRulesByPermCodes(@Param("permCodes") List<String> permCodes);
    
    /**
     * 查询权限关联的字段权限
     * @param permCodes 权限编码列表
     * @param entityType 实体类型
     * @return 字段权限列表
     */
    @Select("<script>" +
            "SELECT " +
            "  fp.field_code, " +
            "  fp.entity_type, " +
            "  fp.field_name, " +
            "  fp.field_alias, " +
            "  fp.perm_type, " +
            "  fp.condition_expression, " +
            "  p.perm_code " +
            "FROM sys_field_permission fp " +
            "JOIN sys_role_permission rp ON fp.role_code = rp.role_code " +
            "JOIN sys_permission p ON rp.perm_code = p.perm_code " +
            "WHERE p.perm_code IN " +
            "  <foreach collection='permCodes' item='permCode' open='(' separator=',' close=')'>" +
            "    #{permCode}" +
            "  </foreach>" +
            "  AND fp.entity_type = #{entityType} " +
            "  AND fp.status = 1 " +
            "  AND fp.is_deleted = 0 " +
            "  AND rp.status = 1 " +
            "  AND rp.is_deleted = 0 " +
            "ORDER BY fp.sort_order" +
            "</script>")
    List<Map<String, Object>> selectFieldPermissionsByPermCodes(
            @Param("permCodes") List<String> permCodes,
            @Param("entityType") String entityType);
    
    /**
     * 查询权限关联的时间权限
     * @param permCodes 权限编码列表
     * @param userNum 用户编号
     * @param roleCodes 角色编码列表
     * @return 时间权限列表
     */
    @Select("<script>" +
            "SELECT " +
            "  tp.time_code, " +
            "  tp.perm_code, " +
            "  tp.allowed_days, " +
            "  tp.start_time, " +
            "  tp.end_time, " +
            "  tp.effective_start, " +
            "  tp.effective_end, " +
            "  tp.timezone, " +
            "  tp.is_recurring, " +
            "  tp.holiday_excluded " +
            "FROM sys_time_permission tp " +
            "WHERE (tp.perm_code IN " +
            "  <foreach collection='permCodes' item='permCode' open='(' separator=',' close=')'>" +
            "    #{permCode}" +
            "  </foreach>" +
            "  OR tp.user_num = #{userNum} " +
            "  OR tp.role_code IN " +
            "  <foreach collection='roleCodes' item='roleCode' open='(' separator=',' close=')'>" +
            "    #{roleCode}" +
            "  </foreach>" +
            ") " +
            "  AND tp.status = 1 " +
            "  AND tp.is_deleted = 0 " +
            "  AND (tp.effective_end IS NULL OR tp.effective_end > CURDATE())" +
            "</script>")
    List<Map<String, Object>> selectTimePermissions(
            @Param("permCodes") List<String> permCodes,
            @Param("userNum") String userNum,
            @Param("roleCodes") List<String> roleCodes);
}
```

### 2.2 统一权限处理Service

```java
package com.example.permission.service;

import com.example.permission.mapper.*;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.DayOfWeek;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.LocalTime;
import java.util.*;
import java.util.stream.Collectors;

@Slf4j
@Service
public class UnifiedPermissionService {
    
    @Autowired
    private UnifiedPermissionMapper unifiedPermissionMapper;
    
    @Autowired
    private SysUserRoleMapper sysUserRoleMapper;
    
    @Autowired
    private SysUserOrgMapper sysUserOrgMapper;
    
    /**
     * 获取用户的完整权限信息（大一统）
     * @param userNum 用户编号
     * @param orgCode 机构编码
     * @return 完整权限信息
     */
    @Transactional(readOnly = true)
    public Map<String, Object> getUserUnifiedPermissions(String userNum, String orgCode) {
        log.info("开始查询用户[{}]在机构[{}]的完整权限信息", userNum, orgCode);
        
        try {
            // 1. 验证用户机构关联
            validateUserOrgAccess(userNum, orgCode);
            
            // 2. 获取用户角色
            List<Map<String, Object>> userRoles = sysUserRoleMapper.selectUserRolesWithDataScope(userNum);
            List<String> roleCodes = userRoles.stream()
                    .map(role -> (String) role.get("role_code"))
                    .collect(Collectors.toList());
            
            // 3. 查询所有权限（大一统查询）
            List<Map<String, Object>> allPermissions = 
                unifiedPermissionMapper.selectAllPermissionsByUserAndOrg(userNum, orgCode);
            
            // 4. 按权限类型分组处理
            Map<Integer, List<Map<String, Object>>> permissionsByType = 
                allPermissions.stream().collect(Collectors.groupingBy(
                    perm -> (Integer) perm.get("perm_type")
                ));
            
            // 5. 提取权限编码
            List<String> permCodes = allPermissions.stream()
                    .map(perm -> (String) perm.get("perm_code"))
                    .distinct()
                    .collect(Collectors.toList());
            
            // 6. 查询关联权限信息
            Map<String, Object> relatedPermissions = queryRelatedPermissions(permCodes, userNum, roleCodes);
            
            // 7. 构建菜单树
            List<Map<String, Object>> menuTree = buildMenuTree(permissionsByType.get(1));
            
            // 8. 构建操作权限集合
            Map<String, Boolean> operationPermissions = buildOperationPermissions(permissionsByType.get(2));
            
            // 9. 构建数据权限规则
            List<Map<String, Object>> dataPermissions = buildDataPermissions(
                    permissionsByType.get(4), 
                    (List<Map<String, Object>>) relatedPermissions.get("dataRules"));
            
            // 10. 构建字段权限
            Map<String, List<Map<String, Object>>> fieldPermissions = buildFieldPermissions(
                    (List<Map<String, Object>>) relatedPermissions.get("fieldPermissions"));
            
            // 11. 构建时间权限
            Map<String, Map<String, Object>> timePermissions = buildTimePermissions(
                    (List<Map<String, Object>>) relatedPermissions.get("timePermissions"));
            
            // 12. 构建返回结果
            Map<String, Object> result = buildUnifiedPermissionResult(
                    userNum, orgCode, userRoles, menuTree, operationPermissions,
                    dataPermissions, fieldPermissions, timePermissions);
            
            log.info("用户[{}]在机构[{}]的完整权限信息查询成功", userNum, orgCode);
            return result;
            
        } catch (Exception e) {
            log.error("查询用户[{}]在机构[{}]的权限信息失败", userNum, orgCode, e);
            throw new RuntimeException("权限查询失败: " + e.getMessage());
        }
    }
    
    /**
     * 验证用户对机构的访问权限
     */
    private void validateUserOrgAccess(String userNum, String orgCode) {
        Map<String, Object> userOrgRelation = 
            sysUserOrgMapper.selectUserOrgRelation(userNum, orgCode);
        if (userOrgRelation == null) {
            throw new RuntimeException("用户[" + userNum + "]无权访问机构[" + orgCode + "]");
        }
    }
    
    /**
     * 查询关联权限信息
     */
    private Map<String, Object> queryRelatedPermissions(List<String> permCodes, 
                                                      String userNum, 
                                                      List<String> roleCodes) {
        Map<String, Object> result = new HashMap<>();
        
        // 数据规则
        List<Map<String, Object>> dataRules = 
            unifiedPermissionMapper.selectDataRulesByPermCodes(permCodes);
        result.put("dataRules", dataRules);
        
        // 字段权限（以用户表为例）
        List<Map<String, Object>> fieldPermissions = 
            unifiedPermissionMapper.selectFieldPermissionsByPermCodes(permCodes, "sys_user");
        result.put("fieldPermissions", fieldPermissions);
        
        // 时间权限
        List<Map<String, Object>> timePermissions = 
            unifiedPermissionMapper.selectTimePermissions(permCodes, userNum, roleCodes);
        result.put("timePermissions", timePermissions);
        
        return result;
    }
    
    /**
     * 构建菜单树
     */
    private List<Map<String, Object>> buildMenuTree(List<Map<String, Object>> menuPermissions) {
        if (menuPermissions == null || menuPermissions.isEmpty()) {
            return new ArrayList<>();
        }
        
        // 按父菜单分组
        Map<String, List<Map<String, Object>>> menusByParent = menuPermissions.stream()
                .collect(Collectors.groupingBy(
                    menu -> (String) menu.get("parent_menu_code"),
                    HashMap::new,
                    Collectors.toList()
                ));
        
        // 构建树结构
        return buildMenuTreeRecursive(null, menusByParent);
    }
    
    private List<Map<String, Object>> buildMenuTreeRecursive(String parentCode,
                                                            Map<String, List<Map<String, Object>>> menusByParent) {
        List<Map<String, Object>> result = new ArrayList<>();
        
        List<Map<String, Object>> children = menusByParent.get(parentCode);
        if (children == null) {
            return result;
        }
        
        for (Map<String, Object> menu : children) {
            Map<String, Object> node = new HashMap<>();
            node.put("menuCode", menu.get("menu_code"));
            node.put("menuName", menu.get("menu_name"));
            node.put("menuType", menu.get("menu_type"));
            node.put("icon", menu.get("icon"));
            node.put("path", menu.get("path"));
            node.put("component", menu.get("component"));
            node.put("url", menu.get("url"));
            node.put("isExternal", menu.get("is_external"));
            node.put("isCache", menu.get("is_cache"));
            node.put("isVisible", menu.get("is_visible"));
            node.put("sortOrder", menu.get("menu_order"));
            node.put("permissionKey", menu.get("perm_key"));
            
            // 递归构建子菜单
            String currentMenuCode = (String) menu.get("menu_code");
            List<Map<String, Object>> subMenus = buildMenuTreeRecursive(currentMenuCode, menusByParent);
            if (!subMenus.isEmpty()) {
                node.put("children", subMenus);
            }
            
            result.add(node);
        }
        
        // 按排序号排序
        result.sort(Comparator.comparingInt(node -> (Integer) node.get("sortOrder")));
        return result;
    }
    
    /**
     * 构建操作权限集合
     */
    private Map<String, Boolean> buildOperationPermissions(List<Map<String, Object>> operationPermissions) {
        Map<String, Boolean> result = new HashMap<>();
        
        if (operationPermissions != null) {
            for (Map<String, Object> perm : operationPermissions) {
                String permKey = (String) perm.get("perm_key");
                Integer authType = (Integer) perm.get("auth_type");
                // auth_type: 1-允许，2-拒绝
                result.put(permKey, authType == 1);
            }
        }
        
        return result;
    }
    
    /**
     * 构建数据权限规则
     */
    private List<Map<String, Object>> buildDataPermissions(List<Map<String, Object>> dataPerms,
                                                          List<Map<String, Object>> dataRules) {
        List<Map<String, Object>> result = new ArrayList<>();
        
        if (dataPerms == null || dataPerms.isEmpty()) {
            return result;
        }
        
        // 按实体类型分组数据权限
        Map<String, List<Map<String, Object>>> permsByEntity = dataPerms.stream()
                .collect(Collectors.groupingBy(
                    perm -> (String) perm.get("entity_type")
                ));
        
        // 按权限编码分组数据规则
        Map<String, List<Map<String, Object>>> rulesByPerm = dataRules.stream()
                .collect(Collectors.groupingBy(
                    rule -> (String) rule.get("perm_code")
                ));
        
        for (String entityType : permsByEntity.keySet()) {
            Map<String, Object> entityPermission = new HashMap<>();
            entityPermission.put("entityType", entityType);
            
            List<Map<String, Object>> entityRules = new ArrayList<>();
            
            for (Map<String, Object> perm : permsByEntity.get(entityType)) {
                String permCode = (String) perm.get("perm_code");
                List<Map<String, Object>> rules = rulesByPerm.get(permCode);
                
                if (rules != null) {
                    for (Map<String, Object> rule : rules) {
                        Map<String, Object> ruleInfo = new HashMap<>();
                        ruleInfo.put("ruleCode", rule.get("rule_code"));
                        ruleInfo.put("ruleName", rule.get("rule_name"));
                        ruleInfo.put("ruleType", rule.get("rule_type"));
                        ruleInfo.put("scopeType", rule.get("scope_type"));
                        ruleInfo.put("includeChildren", rule.get("include_children"));
                        ruleInfo.put("ruleExpression", rule.get("rule_expression"));
                        ruleInfo.put("customSql", rule.get("custom_sql"));
                        ruleInfo.put("isGlobal", rule.get("is_global"));
                        ruleInfo.put("permKey", perm.get("perm_key"));
                        entityRules.add(ruleInfo);
                    }
                } else {
                    // 如果没有特殊规则，使用权限本身的配置
                    Map<String, Object> defaultRule = new HashMap<>();
                    defaultRule.put("ruleCode", "DEFAULT_" + permCode);
                    defaultRule.put("ruleName", "默认数据范围");
                    defaultRule.put("ruleType", 1); // 预定义
                    defaultRule.put("scopeType", perm.get("scope_type"));
                    defaultRule.put("includeChildren", 1);
                    defaultRule.put("permKey", perm.get("perm_key"));
                    entityRules.add(defaultRule);
                }
            }
            
            // 按优先级排序
            entityRules.sort(Comparator.comparingInt(rule -> {
                Integer priority = (Integer) rule.get("rule_priority");
                return priority != null ? priority : 3; // 默认中优先级
            }));
            
            entityPermission.put("rules", entityRules);
            result.add(entityPermission);
        }
        
        return result;
    }
    
    /**
     * 构建字段权限
     */
    private Map<String, List<Map<String, Object>>> buildFieldPermissions(
            List<Map<String, Object>> fieldPermissions) {
        Map<String, List<Map<String, Object>>> result = new HashMap<>();
        
        if (fieldPermissions == null || fieldPermissions.isEmpty()) {
            return result;
        }
        
        // 按实体类型分组
        Map<String, List<Map<String, Object>>> fieldsByEntity = fieldPermissions.stream()
                .collect(Collectors.groupingBy(
                    field -> (String) field.get("entity_type")
                ));
        
        for (String entityType : fieldsByEntity.keySet()) {
            List<Map<String, Object>> entityFields = new ArrayList<>();
            
            for (Map<String, Object> field : fieldsByEntity.get(entityType)) {
                Map<String, Object> fieldInfo = new HashMap<>();
                fieldInfo.put("fieldCode", field.get("field_code"));
                fieldInfo.put("fieldName", field.get("field_name"));
                fieldInfo.put("fieldAlias", field.get("field_alias"));
                fieldInfo.put("permType", field.get("perm_type"));
                fieldInfo.put("conditionExpression", field.get("condition_expression"));
                fieldInfo.put("permCode", field.get("perm_code"));
                entityFields.add(fieldInfo);
            }
            
            // 按字段名去重，保留优先级高的（权限类型数字小的优先级高）
            Map<String, Map<String, Object>> uniqueFields = new HashMap<>();
            for (Map<String, Object> field : entityFields) {
                String fieldName = (String) field.get("field_name");
                Map<String, Object> existing = uniqueFields.get(fieldName);
                
                if (existing == null || ((Integer) field.get("perm_type")) < 
                    ((Integer) existing.get("perm_type"))) {
                    uniqueFields.put(fieldName, field);
                }
            }
            
            result.put(entityType, new ArrayList<>(uniqueFields.values()));
        }
        
        return result;
    }
    
    /**
     * 构建时间权限
     */
    private Map<String, Map<String, Object>> buildTimePermissions(
            List<Map<String, Object>> timePermissions) {
        Map<String, Map<String, Object>> result = new HashMap<>();
        
        if (timePermissions == null || timePermissions.isEmpty()) {
            return result;
        }
        
        // 检查当前时间是否在允许的访问时间内
        LocalDateTime now = LocalDateTime.now();
        LocalDate today = now.toLocalDate();
        LocalTime currentTime = now.toLocalTime();
        DayOfWeek currentDay = now.getDayOfWeek();
        int currentDayNum = currentDay.getValue(); // 1-7, 周一到周日
        
        for (Map<String, Object> timePerm : timePermissions) {
            String permCode = (String) timePerm.get("perm_code");
            
            // 检查生效日期
            Date effectiveStart = (Date) timePerm.get("effective_start");
            Date effectiveEnd = (Date) timePerm.get("effective_end");
            
            if (effectiveStart != null && today.isBefore(effectiveStart.toLocalDate())) {
                continue; // 尚未生效
            }
            if (effectiveEnd != null && today.isAfter(effectiveEnd.toLocalDate())) {
                continue; // 已过期
            }
            
            // 检查星期几
            String allowedDays = (String) timePerm.get("allowed_days");
            if (allowedDays != null) {
                List<String> daysList = Arrays.asList(allowedDays.split(","));
                if (!daysList.contains(String.valueOf(currentDayNum))) {
                    continue; // 今天不允许访问
                }
            }
            
            // 检查时间段
            LocalTime startTime = timePerm.get("start_time") != null ? 
                ((Date) timePerm.get("start_time")).toLocalTime() : null;
            LocalTime endTime = timePerm.get("end_time") != null ? 
                ((Date) timePerm.get("end_time")).toLocalTime() : null;
            
            if (startTime != null && currentTime.isBefore(startTime)) {
                continue; // 未到开始时间
            }
            if (endTime != null && currentTime.isAfter(endTime)) {
                continue; // 已过结束时间
            }
            
            // 检查是否排除节假日（这里需要节假日服务，简单实现）
            boolean isHoliday = isHoliday(today);
            if (timePerm.get("holiday_excluded") != null && 
                (Integer) timePerm.get("holiday_excluded") == 1 && isHoliday) {
                continue; // 节假日排除
            }
            
            // 构建时间权限信息
            Map<String, Object> timeInfo = new HashMap<>();
            timeInfo.put("timeCode", timePerm.get("time_code"));
            timeInfo.put("allowedDays", allowedDays);
            timeInfo.put("startTime", startTime);
            timeInfo.put("endTime", endTime);
            timeInfo.put("effectiveStart", effectiveStart);
            timeInfo.put("effectiveEnd", effectiveEnd);
            timeInfo.put("timezone", timePerm.get("timezone"));
            timeInfo.put("isRecurring", timePerm.get("is_recurring"));
            timeInfo.put("holidayExcluded", timePerm.get("holiday_excluded"));
            timeInfo.put("isValid", true);
            
            result.put(permCode, timeInfo);
        }
        
        return result;
    }
    
    /**
     * 构建统一权限结果
     */
    private Map<String, Object> buildUnifiedPermissionResult(String userNum, String orgCode,
                                                           List<Map<String, Object>> userRoles,
                                                           List<Map<String, Object>> menuTree,
                                                           Map<String, Boolean> operationPermissions,
                                                           List<Map<String, Object>> dataPermissions,
                                                           Map<String, List<Map<String, Object>>> fieldPermissions,
                                                           Map<String, Map<String, Object>> timePermissions) {
        Map<String, Object> result = new HashMap<>();
        
        // 1. 基础信息
        result.put("userNum", userNum);
        result.put("orgCode", orgCode);
        result.put("queryTime", LocalDateTime.now());
        
        // 2. 角色信息
        result.put("userRoles", userRoles);
        
        // 3. 菜单权限
        result.put("menuTree", menuTree);
        
        // 4. 操作权限集合
        result.put("operationPermissions", operationPermissions);
        
        // 5. 数据权限规则
        result.put("dataPermissions", dataPermissions);
        
        // 6. 字段权限
        result.put("fieldPermissions", fieldPermissions);
        
        // 7. 时间权限
        result.put("timePermissions", timePermissions);
        
        // 8. 权限摘要
        Map<String, Object> permissionSummary = new HashMap<>();
        permissionSummary.put("menuCount", countMenuItems(menuTree));
        permissionSummary.put("operationCount", operationPermissions.size());
        permissionSummary.put("dataPermissionEntityCount", dataPermissions.size());
        permissionSummary.put("fieldPermissionEntityCount", fieldPermissions.size());
        permissionSummary.put("timePermissionCount", timePermissions.size());
        result.put("permissionSummary", permissionSummary);
        
        // 9. 权限校验函数（前端可使用）
        result.put("hasPermission", buildPermissionCheckFunctions(operationPermissions));
        
        return result;
    }
    
    /**
     * 统计菜单项数量
     */
    private int countMenuItems(List<Map<String, Object>> menuTree) {
        int count = 0;
        for (Map<String, Object> menu : menuTree) {
            count++;
            if (menu.containsKey("children")) {
                @SuppressWarnings("unchecked")
                List<Map<String, Object>> children = (List<Map<String, Object>>) menu.get("children");
                count += countMenuItems(children);
            }
        }
        return count;
    }
    
    /**
     * 构建权限校验函数
     */
    private Map<String, Object> buildPermissionCheckFunctions(Map<String, Boolean> operationPermissions) {
        Map<String, Object> functions = new HashMap<>();
        
        // 简单校验函数
        functions.put("check", (PermissionChecker) (permKey) -> {
            Boolean hasPerm = operationPermissions.get(permKey);
            return hasPerm != null && hasPerm;
        });
        
        // 批量校验函数
        functions.put("checkAny", (BatchPermissionChecker) (permKeys) -> {
            for (String permKey : permKeys) {
                Boolean hasPerm = operationPermissions.get(permKey);
                if (hasPerm != null && hasPerm) {
                    return true;
                }
            }
            return false;
        });
        
        functions.put("checkAll", (BatchPermissionChecker) (permKeys) -> {
            for (String permKey : permKeys) {
                Boolean hasPerm = operationPermissions.get(permKey);
                if (hasPerm == null || !hasPerm) {
                    return false;
                }
            }
            return true;
        });
        
        return functions;
    }
    
    /**
     * 判断是否为节假日（简化实现，实际应调用节假日服务）
     */
    private boolean isHoliday(LocalDate date) {
        // 这里可以集成第三方节假日API
        // 简化实现：周末为节假日
        DayOfWeek dayOfWeek = date.getDayOfWeek();
        return dayOfWeek == DayOfWeek.SATURDAY || dayOfWeek == DayOfWeek.SUNDAY;
    }
    
    // 函数式接口定义
    @FunctionalInterface
    interface PermissionChecker {
        boolean check(String permKey);
    }
    
    @FunctionalInterface
    interface BatchPermissionChecker {
        boolean check(List<String> permKeys);
    }
}
```

### 2.4 权限缓存策略

```java
@Component
@Slf4j
public class PermissionCacheManager {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    private static final String PERMISSION_CACHE_PREFIX = "perm:";
    private static final long CACHE_EXPIRE_HOURS = 2;
    
    /**
     * 获取缓存的权限信息
     */
    @SuppressWarnings("unchecked")
    public Map<String, Object> getCachedPermissions(String userNum, String orgCode) {
        String cacheKey = buildCacheKey(userNum, orgCode);
        try {
            return (Map<String, Object>) redisTemplate.opsForValue().get(cacheKey);
        } catch (Exception e) {
            log.warn("获取权限缓存失败", e);
            return null;
        }
    }
    
    /**
     * 缓存权限信息
     */
    public void cachePermissions(String userNum, String orgCode, 
                               Map<String, Object> permissions) {
        String cacheKey = buildCacheKey(userNum, orgCode);
        try {
            redisTemplate.opsForValue().set(
                cacheKey, 
                permissions, 
                CACHE_EXPIRE_HOURS, 
                TimeUnit.HOURS
            );
            log.info("用户[{}]在机构[{}]的权限信息已缓存", userNum, orgCode);
        } catch (Exception e) {
            log.error("缓存权限信息失败", e);
        }
    }
    
    /**
     * 清除权限缓存
     */
    public void clearCache(String userNum, String orgCode) {
        String cacheKey = buildCacheKey(userNum, orgCode);
        redisTemplate.delete(cacheKey);
        log.info("用户[{}]在机构[{}]的权限缓存已清除", userNum, orgCode);
    }
    
    private String buildCacheKey(String userNum, String orgCode) {
        return PERMISSION_CACHE_PREFIX + userNum + ":" + orgCode;
    }
}
```

## 四、优势与特点

1. **统一查询**：一次查询获取所有类型权限，减少数据库访问
2. **统一管理**：所有权限通过 `sys_permission` 表统一管理
3. **灵活扩展**：支持7种权限类型，易于扩展新类型
4. **高效缓存**：统一缓存策略，提高权限检查效率
5. **完整集成**：菜单、操作、数据、字段、时间权限一体化
6. **实时生效**：支持时间有效性、条件表达式等动态控制
7. **优先级机制**：支持权限冲突时的优先级处理

## 五、注意事项

1. **性能优化**：大数据量时需考虑分页、索引优化
2. **缓存同步**：权限变更时需及时清除缓存
3. **事务管理**：权限分配需要事务保证一致性
4. **异常处理**：各种权限检查失败时的优雅降级
5. **日志记录**：详细的权限使用日志，便于审计和调试
6. **版本控制**：权限结构变更时考虑版本兼容性

这个大一统权限设计实现了权限的集中管理、统一查询和综合应用，既保证了系统的灵活性，又提高了权限管理的效率和一致性。
